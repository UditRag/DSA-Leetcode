Optimal Solution = derive a general mathematical formula for number of odd numbers between two numbers 

class Solution {
public:
    int countOdds(int low, int high) {
        // 1 to 7 - 1 3 5 7 
        // 2 to 10 - 3 5 7 9 
        // 1 to 10 - 1 3 5 7 9 
        if(low % 2 ==0 && high % 2 == 0){
            return (high-low)/2;
        }
        return (high-low)/2 +1;
    }
};
